using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
using Photon.Pun;
using TMPro;
using UnityEngine.EventSystems;

public class Move_Player : MonoBehaviour
{
    PhotonView view;
    public TextMeshPro nick;

    [Header("Управление камерой")]
    private float xRotation;
    private float sensitivity = 50f;
    private float sensMultiplier = 1f;
    public Camera cam;

    [Header("Прыжок и гравитация")]
    public float jumpSpeed = 8.0F;
    public float gravity = 20.0F;
    private Vector3 moveDirection = Vector3.zero;

    public Vector3 playerScale;

    [Header("Скорость перемещения персонажа")]
    public float baseSpeed = 7f;
    public float runSpeed = 15f;
    private float currentSpeed;

    [Header("Спавн и дестрой объектов.")]
    public GameObject Prefab;
    public GameObject aim;
    Ray spawnRay;
    RaycastHit hit;

    [Header("Инвентарь (Стаки до 32)")]
    public GameObject[] blocks;           
    public int[] blockCounts;             
    public const int MaxStackSize = 32;   
    public GameObject[] cellPanels;       
    public TextMeshProUGUI[] countTexts;  
    public int numPanels;
    public GameObject Torch;
    public GameObject Canvas;

    [Header("Перетаскивание в инвентаре (Мышь)")]
    private GameObject mouseHeldBlock = null;  
    private int mouseHeldCount = 0;            
    public Image cursorItemIcon;               
    public TextMeshProUGUI cursorItemCountText;

    [Header("Настройки дропа")]
    public GameObject customDropPrefab; 

    [Header("Звуки")]
    public AudioClip[] sound;
    public AudioSource music;

    // --- ПЕРЕМЕННЫЕ ДЛЯ РАЗРУШЕНИЯ ---
    private bool isBreaking = false;
    private GameObject targetToBreak = null;
    private float breakTimer = 0f;
    public float breakTime = 1.0f;
    public float breakSoundInterval = 0.3f;
    private float nextSoundTime = 0f;

    // --- МЯГКАЯ ТРЯСКА КАМЕРЫ ---
    [Tooltip("Сила тряски.")]
    public float shakeIntensity = 0.00375f;      
    public float runShakeMultiplier = 1.8f;    
    private const float shakeSpeed = 25f;       
    private float shakeTimer = 0f;              
    private Vector3 shakeOffset = Vector3.zero; 
    private float targetShake = 0f;             
    private float currentShake = 0f;            
    private const float shakeSmooth = 18f;      
    private Vector3 originalCamOffset;  

    void Start()
    {
        view = GetComponent<PhotonView>();
        if (nick != null) nick.text = "Игрок";
        
        numPanels = 0;
        currentSpeed = baseSpeed;

        if (blockCounts == null || blockCounts.Length != blocks.Length)
        {
            blockCounts = new int[blocks.Length];
            for (int i = 0; i < blocks.Length; i++)
            {
                if (blocks[i] != null) blockCounts[i] = 1;
            }
        }

        UpdateAllInventoryUI();

        if (cam != null)
            originalCamOffset = cam.transform.localPosition;
    }

    void Update()
    {
        if (view.IsMine == true)
        {
            cam.gameObject.SetActive(true);
            Canvas.gameObject.SetActive(true);
        }

        if (view.IsMine)
        {
            Cursor.lockState = CursorLockMode.Locked;
            Cursor.visible = false;

            if (ChatManager.isChatActive)
            {
                Cursor.lockState = CursorLockMode.None;
                Cursor.visible = true;
                return;
            }

            HandleSpeed();
            Move();
            
            SpawnObject(); 
            DestroyObject(); 
            Inventory();
            TorchSetActive();
            DropItem(); 
            HandleInventoryMouseClick(); 
            UpdateCursorVisual();        
            esc_menu();
        }
    }

    void HandleSpeed()
    {
        if (Input.GetKey(KeyCode.LeftShift))
        {
            currentSpeed = runSpeed;
        }
        else
        {
            currentSpeed = baseSpeed;
        }
    }

    private void Move()
    {
        CharacterController controller = GetComponent<CharacterController>();
        if (controller == null) return;

        float x = Input.GetAxis("Horizontal");
        float z = Input.GetAxis("Vertical");
        
        Vector3 move = transform.right * x + transform.forward * z;
        move *= currentSpeed;

        if (controller.isGrounded)
        {
            if (Input.GetKey(KeyCode.Space))
            {
                moveDirection.y = jumpSpeed;
            }
            else
            {
                moveDirection.y = 0f;
            }
        }
        
        moveDirection.y -= gravity * Time.deltaTime;

        controller.Move((move + moveDirection) * Time.deltaTime);

        bool isMoving = (move.magnitude > 0.1f);
        ApplyCameraShake(isMoving);
    }

    void ApplyCameraShake(bool isMoving)
    {
        if (cam == null) return;

        float baseShake = shakeIntensity;
        if (Input.GetKey(KeyCode.LeftShift) && isMoving)
        {
            baseShake *= runShakeMultiplier;
        }

        targetShake = isMoving ? baseShake : 0f;
        currentShake = Mathf.Lerp(currentShake, targetShake, Time.deltaTime * shakeSmooth);

        if (currentShake < 0.0001f)
        {
            cam.transform.localPosition = originalCamOffset;
            shakeOffset = Vector3.zero;
            return;
        }

        shakeTimer += Time.deltaTime * shakeSpeed;
        float sX = Mathf.Sin(shakeTimer) * currentShake;
        float sY = Mathf.Cos(shakeTimer * 1.3f) * currentShake * 0.6f;
        float sZ = Mathf.Sin(shakeTimer * 0.7f) * currentShake * 0.4f;

        shakeOffset = new Vector3(sX, sY, sZ);
        cam.transform.localPosition = originalCamOffset + shakeOffset;
    }

    public void SpawnObject()
    {
        if (Input.GetKeyUp(KeyCode.Mouse1))
        {
            if (mouseHeldBlock != null) return;

            if (numPanels == 9 || blocks == null || numPanels >= blocks.Length || blocks[numPanels] == null) 
                return;

            if (blockCounts[numPanels] <= 0)
                return;

            spawnRay = cam.ScreenPointToRay(Input.mousePosition);

            if (Physics.Raycast(spawnRay, out hit, 10.0f))
            {
                bool spawned = false;
                Vector3 spawnPos = Vector3.zero;

                if (hit.transform.tag == "floor")
                {
                    spawnPos = hit.point;
                    spawned = true;
                }
                else if (hit.transform.gameObject.name == "Trigger Y")
                {
                    spawnPos = hit.transform.position + new Vector3(0, 0.459f, 0);
                    spawned = true;
                }
                else if (hit.transform.gameObject.name == "Trigger -Y") 
                {
                    spawnPos = hit.transform.position + new Vector3(0, -0.459f, 0);
                    spawned = true;
                }
                else if (hit.transform.gameObject.name == "Trigger X")
                {
                    spawnPos = hit.transform.position + new Vector3(0.459f, 0, 0);
                    spawned = true;
                }
                else if (hit.transform.gameObject.name == "Trigger -X")
                {
                    spawnPos = hit.transform.position + new Vector3(-0.459f, 0, 0);
                    spawned = true;
                }
                else if (hit.transform.gameObject.name == "Trigger Z")
                {
                    spawnPos = hit.transform.position + new Vector3(0, 0, 0.459f);
                    spawned = true;
                }
                else if (hit.transform.gameObject.name == "Trigger -Z")
                {
                    spawnPos = hit.transform.position + new Vector3(0, 0, -0.459f);
                    spawned = true;
                }

                if (spawned)
                {
                    GameObject blockToSpawn = blocks[numPanels];
                    GameObject spawnedObj = PhotonNetwork.Instantiate(blockToSpawn.name, spawnPos, Quaternion.identity);

                    PlacedBlockIdentifier identifier = spawnedObj.GetComponent<PlacedBlockIdentifier>();
                    if (identifier == null)
                    {
                        identifier = spawnedObj.AddComponent<PlacedBlockIdentifier>();
                    }
                    identifier.originalPrefab = blockToSpawn;

                    PlaySound(0);

                    blockCounts[numPanels]--;
                    if (blockCounts[numPanels] <= 0)
                    {
                        blocks[numPanels] = null;
                        blockCounts[numPanels] = 0;
                    }

                    UpdateInventorySlotUI(numPanels);
                }
            }
        }
    }

    public void DestroyObject()
    {
        if (Input.GetMouseButton(0))
        {
            if (!isBreaking)
            {
                spawnRay = cam.ScreenPointToRay(Input.mousePosition);

                if (Physics.Raycast(spawnRay, out hit, 10.0f))
                {
                    if (hit.transform.GetComponentInParent<DroppedItem>() != null || hit.transform.GetComponent<DroppedItem>() != null)
                    {
                        return;
                    }

                    GameObject candidate = null;

                    if (hit.transform.CompareTag("trigger"))
                    {
                        if (hit.transform.parent != null)
                            candidate = hit.transform.parent.gameObject;
                    }
                    else if (hit.transform.CompareTag("Block"))
                    {
                        candidate = hit.transform.gameObject;
                    }
                    else if (hit.transform.GetComponentInParent<PhotonView>() != null)
                    {
                        PhotonView pvHit = hit.transform.GetComponentInParent<PhotonView>();
                        if (!pvHit.CompareTag("Player"))
                        {
                            candidate = pvHit.gameObject;
                        }
                    }

                    if (candidate != null)
                    {
                        isBreaking = true;
                        targetToBreak = candidate;
                        breakTimer = 0f;
                        nextSoundTime = Time.time + breakSoundInterval;
                    }
                }
            }
            else if (targetToBreak != null)
            {
                if (targetToBreak.GetComponent<DroppedItem>() != null || targetToBreak.GetComponentInChildren<DroppedItem>() != null)
                {
                    isBreaking = false;
                    targetToBreak = null;
                    breakTimer = 0f;
                    return;
                }

                breakTimer += Time.deltaTime;

                if (Time.time >= nextSoundTime)
                {
                    PlaySound(2); 
                    nextSoundTime = Time.time + breakSoundInterval;
                }

                if (breakTimer >= breakTime)
                {
                    GameObject blockPrefabToReturn = null;
                    PlacedBlockIdentifier identifier = targetToBreak.GetComponent<PlacedBlockIdentifier>();
                    if (identifier != null && identifier.originalPrefab != null)
                    {
                        blockPrefabToReturn = identifier.originalPrefab;
                    }
                    
                    if (blockPrefabToReturn == null)
                    {
                        string cleanName = targetToBreak.name.Replace("(Clone)", "").Trim();
                        foreach (GameObject b in blocks)
                        {
                            if (b != null && b.name == cleanName)
                            {
                                blockPrefabToReturn = b;
                                break;
                            }
                        }
                    }

                    if (blockPrefabToReturn == null)
                    {
                        foreach (GameObject b in blocks)
                        {
                            if (b != null)
                            {
                                blockPrefabToReturn = b;
                                break;
                            }
                        }
                    }

                    bool added = false;
                    if (blockPrefabToReturn != null)
                    {
                        added = ReturnBlockToInventory(numPanels, blockPrefabToReturn);

                        if (!added)
                        {
                            for (int i = 0; i < blocks.Length; i++)
                            {
                                if (blocks[i] == blockPrefabToReturn && blockCounts[i] < MaxStackSize)
                                {
                                    added = ReturnBlockToInventory(i, blockPrefabToReturn);
                                    break;
                                }
                            }
                        }

                        if (!added)
                        {
                            for (int i = 0; i < blocks.Length; i++)
                            {
                                if (blocks[i] == null)
                                {
                                    added = ReturnBlockToInventory(i, blockPrefabToReturn);
                                    break;
                                }
                            }
                        }
                    }

                    if (added)
                    {
                        PhotonView targetPV = targetToBreak.GetComponent<PhotonView>();
                        if (targetPV != null)
                        {
                            if (!targetPV.IsMine)
                            {
                                targetPV.TransferOwnership(PhotonNetwork.LocalPlayer);
                            }
                            PhotonNetwork.Destroy(targetPV.gameObject);
                        }
                        PlaySound(1); 
                    }

                    isBreaking = false;
                    targetToBreak = null;
                    breakTimer = 0f;
                    nextSoundTime = 0f;
                }
            }
        }
        else
        {
            isBreaking = false;
            targetToBreak = null;
            breakTimer = 0f;
            nextSoundTime = 0f;
        }
    }

    public void DropItem()
    {
        if (Input.GetKeyDown(KeyCode.Q))
        {
            if (numPanels == 9 || blocks == null || numPanels >= blocks.Length || blocks[numPanels] == null) 
                return;

            if (blockCounts[numPanels] <= 0)
                return;

            if (cam != null)
            {
                GameObject blockToDrop = blocks[numPanels];
                
                Vector3 spawnPos = cam.transform.position + cam.transform.forward * 1.5f;

                RaycastHit groundHit;
                if (Physics.Raycast(cam.transform.position + cam.transform.forward * 1.5f, Vector3.down, out groundHit, 5f))
                {
                    spawnPos = groundHit.point + new Vector3(0, 0.2f, 0); 
                }

                GameObject prefabToSpawn = (customDropPrefab != null) ? customDropPrefab : blockToDrop;

                GameObject droppedObj = PhotonNetwork.Instantiate(prefabToSpawn.name, spawnPos, Quaternion.identity);

                DroppedItem itemScript = droppedObj.GetComponent<DroppedItem>();
                if (itemScript == null)
                {
                    itemScript = droppedObj.AddComponent<DroppedItem>();
                }
                
                itemScript.blockIndex = numPanels;
                itemScript.originalBlockPrefab = blockToDrop; 

                blockCounts[numPanels]--;
                if (blockCounts[numPanels] <= 0)
                {
                    blocks[numPanels] = null;
                    blockCounts[numPanels] = 0;
                }

                UpdateInventorySlotUI(numPanels);
                PlaySound(0);
            }
        }
    }

    public bool ReturnBlockToInventory(int index, GameObject prefabToReturn)
    {
        if (index < 0 || index >= blocks.Length) return false;

        if (blockCounts == null || blockCounts.Length != blocks.Length)
        {
            blockCounts = new int[blocks.Length];
        }

        if ((blocks[index] == null || blocks[index] == prefabToReturn) && blockCounts[index] < MaxStackSize)
        {
            blocks[index] = prefabToReturn;
            blockCounts[index]++;
            
            UpdateInventorySlotUI(index);
            PlaySound(0);
            return true;
        }
        return false;
    }

    void HandleInventoryMouseClick()
    {
        if (Input.GetMouseButtonDown(0))
        {
            for (int i = 0; i < cellPanels.Length; i++)
            {
                if (cellPanels[i] != null)
                {
                    RectTransform rectTransform = cellPanels[i].GetComponent<RectTransform>();
                    if (RectTransformUtility.RectangleContainsScreenPoint(rectTransform, Input.mousePosition, cam))
                    {
                        OnClickInventorySlot(i);
                        break;
                    }
                }
            }
        }
    }

    void OnClickInventorySlot(int index)
    {
        if (mouseHeldBlock == null)
        {
            if (blocks[index] != null && blockCounts[index] > 0)
            {
                mouseHeldBlock = blocks[index];
                mouseHeldCount = blockCounts[index];

                blocks[index] = null;
                blockCounts[index] = 0;

                UpdateInventorySlotUI(index);
                PlaySound(0);
            }
        }
        else
        {
            if (blocks[index] == null)
            {
                blocks[index] = mouseHeldBlock;
                blockCounts[index] = mouseHeldCount;

                mouseHeldBlock = null;
                mouseHeldCount = 0;

                UpdateInventorySlotUI(index);
                PlaySound(0);
            }
            else if (blocks[index] == mouseHeldBlock)
            {
                int total = blockCounts[index] + mouseHeldCount;
                if (total <= MaxStackSize)
                {
                    blockCounts[index] = total;
                    mouseHeldBlock = null;
                    mouseHeldCount = 0;
                }
                else
                {
                    blockCounts[index] = MaxStackSize;
                    mouseHeldCount = total - MaxStackSize;
                }

                UpdateInventorySlotUI(index);
                PlaySound(0);
            }
            else
            {
                GameObject tempBlock = blocks[index];
                int tempCount = blockCounts[index];

                blocks[index] = mouseHeldBlock;
                blockCounts[index] = mouseHeldCount;

                mouseHeldBlock = tempBlock;
                mouseHeldCount = tempCount;

                UpdateInventorySlotUI(index);
                PlaySound(0);
            }
        }
    }

    void UpdateCursorVisual()
    {
        if (cursorItemIcon != null)
        {
            if (mouseHeldBlock != null && mouseHeldCount > 0)
            {
                cursorItemIcon.gameObject.SetActive(true);
                cursorItemIcon.transform.position = Input.mousePosition;

                if (cursorItemCountText != null)
                {
                    cursorItemCountText.text = (mouseHeldCount > 1) ? mouseHeldCount.ToString() : "";
                }
            }
            else
            {
                cursorItemIcon.gameObject.SetActive(false);
                if (cursorItemCountText != null) cursorItemCountText.text = "";
            }
        }
    }

    void UpdateAllInventoryUI()
    {
        for (int i = 0; i < cellPanels.Length; i++)
        {
            UpdateInventorySlotUI(i);
        }
    }

    void UpdateInventorySlotUI(int index)
    {
        if (index < 0 || index >= cellPanels.Length || cellPanels[index] == null) return;

        if (cellPanels[index].transform.childCount > 0)
        {
            Transform iconChild = cellPanels[index].transform.GetChild(0);
            if (blocks[index] != null && blockCounts[index] > 0)
            {
                iconChild.gameObject.SetActive(true);
            }
            else
            {
                iconChild.gameObject.SetActive(false);
            }
        }

        if (countTexts != null && index < countTexts.Length && countTexts[index] != null)
        {
            countTexts[index].text = (blockCounts[index] > 0) ? blockCounts[index].ToString() : "";
        }
    }

    void PlaySound(int index)
    {
        if (music != null && sound != null && index < sound.Length)
        {
            music.clip = sound[index];
            music.Play();
        }
    }

    public void Inventory()
    {
        if (Input.GetKeyDown(KeyCode.Alpha1)) SetPanel(0);
        else if (Input.GetKeyDown(KeyCode.Alpha2)) SetPanel(1);
        else if (Input.GetKeyDown(KeyCode.Alpha3)) SetPanel(2);
        else if (Input.GetKeyDown(KeyCode.Alpha4)) SetPanel(3);
        else if (Input.GetKeyDown(KeyCode.Alpha5)) SetPanel(4);
        else if (Input.GetKeyDown(KeyCode.Alpha6)) SetPanel(5);
        else if (Input.GetKeyDown(KeyCode.Alpha7)) SetPanel(6);
        else if (Input.GetKeyDown(KeyCode.Alpha8)) SetPanel(7);
    }

    void SetPanel(int index)
    {
        if (index < 0 || index > 8) return;

        if (numPanels >= 0 && numPanels < cellPanels.Length && cellPanels[numPanels] != null)
            cellPanels[numPanels].GetComponent<Image>().color = Color.gray;

        numPanels = index;

        if (numPanels >= 0 && numPanels < cellPanels.Length && cellPanels[numPanels] != null)
            cellPanels[numPanels].GetComponent<Image>().color = Color.white;
    }

    public void TorchSetActive()
    {
        if (numPanels == 9)
        {
            if (Torch != null) Torch.SetActive(true);
        }
        else
        {
            if (Torch != null) Torch.SetActive(false);
        }
    }

    void esc_menu()
    {
        if (Input.GetKeyUp(KeyCode.Escape))
        {
            Cursor.lockState = CursorLockMode.Confined;
            PhotonNetwork.Disconnect();
            PhotonNetwork.LoadLevel("LoadingScene");
        }
    }
}

public class PlacedBlockIdentifier : MonoBehaviour
{
    public GameObject originalPrefab;
}
