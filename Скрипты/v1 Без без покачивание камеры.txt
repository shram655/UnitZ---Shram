using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
using Photon.Pun;
using TMPro;

public class Move_Player : MonoBehaviour
{
PhotonView view;
public TextMeshPro nick;

[Header("Управление камерой")]
private float xRotation;
private float sensitivity = 50f;
private float sensMultiplier = 1f;
public Camera cam;

[Header("Прыжок")]
public float jumpSpeed = 8.0F;
public float gravity = 20.0F;
private Vector3 moveDirection = Vector3.zero;

public Vector3 playerScale;

[Header("Скорость перемещения персонажа")]
public float speed = 7f;

[Header("Спавн и дестрой объектов.")]
public GameObject Prefab;
public GameObject aim;
Ray spawnRay;
RaycastHit hit;

[Header("Инвентарь")]
public GameObject[] blocks;
public GameObject[] cellPanels;
public int numPanels;
public GameObject Torch;
public GameObject Canvas;

[Header("Звуки")]
public AudioClip[] sound;
public AudioSource music;

// --- НОВЫЕ ПЕРЕМЕННЫЕ ДЛЯ РАЗРУШЕНИЯ ---
private bool isBreaking = false; // ломаем ли мы сейчас блок
private GameObject targetToBreak = null; // какой блок ломаем
private float breakTimer = 0f; // сколько уже держим (до полного разрушения)
public float breakTime = 1.0f; // время разрушения (в секундах)

// --- ЗВУКИ РАЗРУШЕНИЯ (СТУКИ) ---
public float breakSoundInterval = 0.3f; // интервал между «ударами»
private float nextSoundTime = 0f; // когда должен быть следующий звук

void Start()
{
view = GetComponent<PhotonView>();
nick.GetComponent<TextMeshPro>().text = "Игрок";
numPanels = 0;
}

void Update()
{
if (view.IsMine == true)
{
cam.gameObject.SetActive(true);
Canvas.gameObject.SetActive(true);
}

if (view.IsMine)
{
Cursor.lockState = CursorLockMode.Locked;
Move();
Shift();
SpawnObject(); // ПКМ — ставить
DestroyObject(); // ЛКМ — ломать с задержкой и звуками
Inventory();
TorchSetActive();
esc_menu();
}
}


// --- СТАВИТЬ БЛОК — НА ПКМ (Mouse1) ---
public void SpawnObject()
{
if (Input.GetKeyUp(KeyCode.Mouse1))
{
spawnRay = cam.ScreenPointToRay(Input.mousePosition);

if (Physics.Raycast (spawnRay, out hit, 10.0f))
{
if (numPanels == 9)
{
return;
}
else if (hit.transform.tag == "floor")
{
PhotonNetwork.Instantiate(blocks[numPanels].name, hit.point, Quaternion.identity);
if (music != null && sound != null && sound.Length > 0)
{
music.clip = sound[0];
music.Play();
}
}
else if(hit.transform.gameObject.name == "Trigger Y")
{
PhotonNetwork.Instantiate(blocks[numPanels].name, hit.point = new Vector3(hit.transform.position.x, hit.transform.position.y, hit.transform.position.z) + new Vector3(0, 0.459f, 0), Quaternion.identity);
if (music != null && sound != null && sound.Length > 0)
{
music.clip = sound[0];
music.Play();
}
}
else if(hit.transform.gameObject.name == "Trigger -Y ")
{
PhotonNetwork.Instantiate(blocks[numPanels].name, hit.point = new Vector3(hit.transform.position.x, hit.transform.position.y, hit.transform.position.z) + new Vector3(0, -0.459f, 0), Quaternion.identity);
if (music != null && sound != null && sound.Length > 0)
{
music.clip = sound[0];
music.Play();
}
}
else if (hit.transform.gameObject.name == "Trigger X")
{
PhotonNetwork.Instantiate(blocks[numPanels].name, hit.point = new Vector3(hit.transform.position.x, hit.transform.position.y, hit.transform.position.z) + new Vector3(0.459f, 0, 0), Quaternion.identity);
if (music != null && sound != null && sound.Length > 0)
{
music.clip = sound[0];
music.Play();
}
}
else if (hit.transform.gameObject.name == "Trigger -X")
{
PhotonNetwork.Instantiate(blocks[numPanels].name, hit.point = new Vector3(hit.transform.position.x, hit.transform.position.y, hit.transform.position.z) + new Vector3(-0.459f, 0, 0), Quaternion.identity);
if (music != null && sound != null && sound.Length > 0)
{
music.clip = sound[0];
music.Play();
}
}
else if (hit.transform.gameObject.name == "Trigger Z")
{
PhotonNetwork.Instantiate(blocks[numPanels].name, hit.point = new Vector3(hit.transform.position.x, hit.transform.position.y, hit.transform.position.z) + new Vector3(0, 0, 0.459f), Quaternion.identity);
if (music != null && sound != null && sound.Length > 0)
{
music.clip = sound[0];
music.Play();
}
}
else if (hit.transform.gameObject.name == "Trigger -Z")
{
PhotonNetwork.Instantiate(blocks[numPanels].name, hit.point = new Vector3(hit.transform.position.x, hit.transform.position.y, hit.transform.position.z) + new Vector3(0, 0, -0.459f), Quaternion.identity);
if (music != null && sound != null && sound.Length > 0)
{
music.clip = sound[0];
music.Play();
}
}
}
}
}

// --- ЛОМАТЬ БЛОК — НА ЛКМ (Mouse0) С ЗАДЕРЖКОЙ И СТУКАМИ ---
public void DestroyObject()
{
// Если мы уже ломаем какой-то блок — продолжаем таймер и звуки
if (isBreaking && targetToBreak != null)
{
breakTimer += Time.deltaTime;

// Логика «стуков» во время разрушения
if (Time.time >= nextSoundTime)
{
// Звук удара (индекс 2 в массиве sound, например)
if (music != null && sound != null && sound.Length > 2)
{
music.clip = sound[2];
music.Play();
}

nextSoundTime = Time.time + breakSoundInterval;
}

// Если набрали нужное время — уничтожаем
if (breakTimer >= breakTime)
{
PhotonView targetPV = targetToBreak.GetComponent<PhotonView>();
if (targetPV != null)
{
PhotonNetwork.Destroy(targetPV.gameObject);
}

// Финальный звук разрушения (индекс 1)
if (music != null && sound != null && sound.Length > 1)
{
music.clip = sound[1];
music.Play();
}

// Сбрасываем состояние
isBreaking = false;
targetToBreak = null;
breakTimer = 0f;
nextSoundTime = 0f;
}
return; // не делаем новый Raycast, пока ломаем текущий блок
}

// Если не ломаем — проверяем нажатие ЛКМ
if (Input.GetKeyDown(KeyCode.Mouse0))
{
spawnRay = cam.ScreenPointToRay(Input.mousePosition);

if (Physics.Raycast(spawnRay, out hit, 10.0f))
{
GameObject candidate = null;

// Ищем, что именно хотим сломать (по тегу или по родителю)
if (hit.transform.CompareTag("trigger"))
{
candidate = hit.transform.parent.gameObject;
}
else if (hit.transform.CompareTag("Block"))
{
candidate = hit.transform.gameObject;
}

if (candidate != null)
{
// Начинаем ломать
isBreaking = true;
targetToBreak = candidate;
breakTimer = 0f;
nextSoundTime = Time.time + breakSoundInterval; // первый удар через интервал
}
}
}
else if (Input.GetKeyUp(KeyCode.Mouse0))
{
// Отпустили ЛКМ раньше времени — сбрасываем
isBreaking = false;
targetToBreak = null;
breakTimer = 0f;
nextSoundTime = 0f;
}
}

// ТОЛЬКО клавиши 1–8 (и 9 для факела) — колёсико полностью убрано
public void Inventory()
{
if (Input.GetKeyDown(KeyCode.Alpha1)) SetPanel(0);
else if (Input.GetKeyDown(KeyCode.Alpha2)) SetPanel(1);
else if (Input.GetKeyDown(KeyCode.Alpha3)) SetPanel(2);
else if (Input.GetKeyDown(KeyCode.Alpha4)) SetPanel(3);
else if (Input.GetKeyDown(KeyCode.Alpha5)) SetPanel(4);
else if (Input.GetKeyDown(KeyCode.Alpha6)) SetPanel(5);
else if (Input.GetKeyDown(KeyCode.Alpha7)) SetPanel(6);
else if (Input.GetKeyDown(KeyCode.Alpha8)) SetPanel(7);
}

void SetPanel(int index)
{
if (index < 0 || index > 8) return;

if (numPanels >= 0 && numPanels < cellPanels.Length)
cellPanels[numPanels].GetComponent<Image>().color = Color.gray;

numPanels = index;

if (numPanels >= 0 && numPanels < cellPanels.Length)
cellPanels[numPanels].GetComponent<Image>().color = Color.white;
}

public void TorchSetActive()
{
if (numPanels == 9)
{
Torch.SetActive(true);
}
else
{
Torch.SetActive(false);
}
}

public void esc_menu()
{
if (Input.GetKeyUp(KeyCode.Escape))
{
Cursor.lockState = CursorLockMode.Confined;
PhotonNetwork.Disconnect();
PhotonNetwork.LoadLevel("LoadingScene");
}
}

private void Move()
{
CharacterController controller = GetComponent<CharacterController>();

if (controller.isGrounded)
{
moveDirection = new Vector3(Input.GetAxis("Horizontal"), 0, Input.GetAxis("Vertical"));
moveDirection = transform.TransformDirection(moveDirection);
moveDirection *= speed;
if (Input.GetKey(KeyCode.W))
{
transform.localPosition += transform.forward * speed * Time.deltaTime;
}
if (Input.GetKey(KeyCode.S))
{
transform.localPosition += -transform.forward * speed * Time.deltaTime;
}
if (Input.GetKey(KeyCode.A))
{
transform.localPosition += -transform.right * speed * Time.deltaTime;
}
if (Input.GetKey(KeyCode.D))
{
transform.localPosition += transform.right * speed * Time.deltaTime;
}
if (Input.GetKey(KeyCode.Space))
{
moveDirection.y = jumpSpeed;
}
}
moveDirection.y -= gravity * Time.deltaTime;
controller.Move(moveDirection * Time.deltaTime);
}

private void Shift()
{
if (Input.GetKey(KeyCode.LeftShift))
{
speed = 15f;
}
else
{
speed = 7f;
}
}
}
